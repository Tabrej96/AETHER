2. MCP Tools (20 tools)

4. TypeScript Types

Key types used across the AETHER engine. All types are in src/core/types.ts and program-specific modules.

Core types

// Program lifecycle states (DISCOVER → INSTALL → VALIDATE → REGISTER → EXECUTE → MONITOR → UPDATE → REMOVE)
type ProgramStatus = 'discovered' | 'installed' | 'validated' | 'registered' | 'executing' | 'monitoring' | 'updating' | 'removed';

// Capability registration
interface Capability {
  id: string;                    // e.g. "cap-p01-development-intelligence"
  name: string;
  version: string;
  status: ProgramStatus;
  provenance: ProvenanceRecord;
  testResults: TestResult[];
  license: string;
}

// Provenance record (every capability has one)
interface ProvenanceRecord {
  upstreamRepo?: string;         // Inspiration source (if any)
  upstreamLicense?: string;      // License of upstream source
  adaptationNotes: string;       // What was extracted and reimplemented
  isFirstParty: boolean;         // true for P10-P21 (no external repo studied)
}

// Test result
interface TestResult {
  suite: string;
  passed: number;
  failed: number;
  flakeFree: boolean;
}

// Measurement record (no fabricated numbers)
interface MeasurementRecord {
  metric: string;
  value: number | string;
  unit?: string;
  status: 'MEASURED' | 'UNMEASURED' | 'UNAVAILABLE';
  reproductionCommand?: string;
}

Program engine interface

// Every program engine implements this contract
interface ProgramEngine {
  readonly id: string;
  readonly name: string;
  readonly version: string;

  // Lifecycle
  discover(): Promise<DiscoveryResult>;
  install(): Promise<InstallResult>;
  validate(): Promise<ValidationResult>;
  register(registry: IntegrationRegistry): Promise<void>;
  execute(input: unknown): Promise<ExecutionResult>;
  monitor(): Promise<MonitorResult>;
  update(version: string): Promise<UpdateResult>;
  remove(): Promise<void>;
}

// Engine result (deterministic, structured)
interface ExecutionResult {
  ok: boolean;
  output: unknown;
  errors: string[];
  metadata: Record<string, unknown>;
  verificationHash: string;     // SHA-256 of output for audit
}

Money types (P10/P16)

// P10 Money primitive — integer paise, never binary floats
type Paise = number & { readonly __brand: 'Paise' };
type Money = {
  paise: Paise;
  currency: 'INR' | 'USD' | 'EUR' | 'GBP';
  source: string;               // Provenance: rate table + effective date
  effectiveDate: string;        // ISO date
};

// P16 Calculation types
interface Commission {
  gross: Paise;
  rate: number;                 // percentage, e.g. 0.05 = 5%
  effectiveDate: string;
  net: Paise;                   // gross - commission
}

interface EMI {
  principal: Paise;
  annualRate: number;           // percentage, e.g. 8 = 8%
  months: number;
  monthlyPaise: Paise;          // BigInt closed form
  totalPaise: Paise;
}

Browser types (P05/P07)

// P05 Browser engine
interface BrowserSnapshot {
  strategy: 'DETERMINISTIC_DOM' | 'ARIA_ROLE' | 'A11Y_TREE' | 'VISION_ESCALATION';
  tokensUsed: number;

5. Integration Patterns

Pattern 1: CLI integration

The simplest integration: call AETHER as a subprocess from your own code.

// Node.js subprocess example
import { spawn } from 'child_process';

function runAether(command, args = []) {
  return new Promise((resolve, reject) => {
    const proc = spawn('npx', ['tsx', 'bin/aether.ts', command, ...args], {
      cwd: '/path/to/AETHER',
      stdio: ['pipe', 'pipe', 'pipe']
    });
    let stdout = '', stderr = '';
    proc.stdout.on('data', d => stdout += d);
    proc.stderr.on('data', d => stderr += d);
    proc.on('close', code => {
      if (code === 0) resolve(JSON.parse(stdout));
      else reject(new Error(stderr || `Exit ${code}`));
    });
  });
}

// Usage
const programs = await runAether('programs');
const result = await runAether('deep', ['--no-browser']);

Pattern 2: MCP integration

For AI coding agents: register AETHER as an MCP server.

// Claude Code: .claude/settings.json
{
  "mcpServers": {
    "aether": {
      "command": "npx",
      "args": ["tsx", "bin/aether.ts", "mcp"],
      "cwd": "/path/to/AETHER"
    }
  }
}

// Then use tools in conversation:
// <UseMcp>
//   <tool>aether_deep_intel</tool>
//   <args>{ "task": "validate invoice #1234" }</args>
// </UseMcp>

Pattern 3: Library import

For TypeScript projects that want to import AETHER engines directly.

// After npm install aether-harness
import { DeepIntelligenceEngine } from 'aether-harness/src/core/intel/deepExecution';
import { FullstackEngine } from 'aether-harness/src/core/fullstack';

const deep = new DeepIntelligenceEngine();
const result = await deep.execute({
  task: 'Compute GST invoice',
  invoice: { /* ... */ }
});

if (result.ok) {
  console.log('Paise total:', result.output.grandTotalPaise);
  console.log('Tax breakdown:', result.output.taxBreakdown);
}

Pattern 4: Harness install

Install AETHER agents and skills into your existing editor/harness.

// Install all packs into Claude Code
npx tsx bin/aether.ts install --all --target claude-code --write .

// Install specific packs into Cursor
npx tsx bin/aether.ts install \
  --target cursor \
  --packs aws-cloud-vps,design-ui,finance-intel \
  --mcp \
  --write .

// Verify what was installed
cat .aether/owned-files.json

Pattern 5: Live provider (optional)

Drive AETHER with a real model for the autonomous runtime.

# .env
AETHER_LIVE_PROVIDER_URL="https://api.anthropic.com/v1"
AETHER_LIVE_PROVIDER_KEY="sk-ant-..."
AETHER_LIVE_PROVIDER_MODEL="claude-sonnet-4-20250514"
AETHER_LIVE_PROVIDER_VENDOR="anthropic"
AETHER_LIVE_PROVIDER_IN_USD_PER_M="3"
AETHER_LIVE_PROVIDER_OUT_USD_PER_M="15"

# Run with live model
npx tsx bin/aether.ts runtime --live
Important: Without AETHER_LIVE_PROVIDER_* keys, every AETHER engine runs deterministically and offline. The live provider is only needed for the model-driven execution paths. All benchmarks, tests, and most engines work without it.
